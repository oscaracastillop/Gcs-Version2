CREATE PROCEDURE [dbo].[SP_BuscarRolUsuario] @IdUser varchar(255), @Resultado varchar(255) OUTPUT
AS
BEGIN
	DECLARE		@Rol varchar(255)
		SELECT	@Rol = R.Nombre 
		FROM	Rol R,
				Usuario U
		WHERE	R.Id = U.IdRol
				and U.IdUser = @IdUser
	
	SET @Resultado = @Rol
	SELECT @Resultado
END

CREATE PROCEDURE [dbo].[SP_CrearUsuario]

				@NombreUsuarioLogin varchar(255),

BEGIN

			INSERT INTO Permiso_SubMenu (IdSubMenu, IdUsuario, Permiso, Crear, Editar, Eliminar) VALUES (43,@IdUserCreado,1,1,1,1)
			INSERT INTO Permiso_SubMenu (IdSubMenu, IdUsuario, Permiso, Crear, Editar, Eliminar) VALUES (44,@IdUserCreado,1,0,0,0)

CREATE PROCEDURE [dbo].[SP_ActualizarUsuario]	@IdUser varchar(255),  
												@IdUsuarioLogin int, 			 
												@Usuario varchar(255), 
												@Password varchar(255),
												@Email varchar(255), 				  
												@NombreUsuarioLogin varchar(255),
												@FechaVigencia varchar(255),
												@IdEstado int,			
												@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@UsuarioMD5 VARCHAR(255),
		@PasswordMD5 VARCHAR(255),
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)
	
SET @Accion = 'Editar'
SET @SubMenu = '/Configuracion/Usuario'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	IF(exists(select * from Usuario where Usuario = @Usuario AND Email = @Email AND Id != @IdUsuarioLogin))  
	BEGIN  
		SET @Resultado = 'Error*El Usuario ' +  @Usuario +' ya existe.'
		RETURN  
	END
ELSE
	BEGIN
		DECLARE @ValidarPassword VARCHAR(255)
		SELECT @ValidarPassword = Password FROM Usuario WHERE Id = @IdUsuarioLogin
		SET @UsuarioMD5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Usuario ), 2)
		IF @Password = @ValidarPassword
			BEGIN
				UPDATE	Usuario 
				SET		Email = @Email, 
						IdUser = @UsuarioMD5,
						Usuario = @Usuario,
						Nombre = @NombreUsuarioLogin,
						FechaVigencia = @FechaVigencia + ' 23:59',
						IdEstado = @IdEstado
				WHERE	Id = @IdUsuarioLogin
			END
		ELSE
			BEGIN
			SET @PasswordMD5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Password ), 2)
				UPDATE	Usuario 
				SET		Password = @PasswordMD5, 							 
						Email = @Email, 
						Usuario = @Usuario,
						Nombre = @NombreUsuarioLogin,
						IdUser = @UsuarioMD5,
						FechaVigencia = @FechaVigencia + ' 23:59',
						IdEstado = @IdEstado
				WHERE	Id = @IdUsuarioLogin
			END		
		SET @Resultado = 'OK*El Usuario ' +  @Usuario + ' se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_EliminarUsuario] @IdUser varchar(255), @IdUsuario int, @Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuarioCrea INT, 
		@NombreUsuario varchar(255),
		@Accion varchar(255), 
		@SubMenu varchar(255)

SET @Accion = 'Eliminar'
SET @SubMenu = '/Configuracion/Usuario'

SELECT @IdUsuarioCrea = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @NombreUsuario = Nombre FROM Usuario WHERE Id = @IdUsuario
SELECT	@Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuarioCrea, @SubMenu)	

IF @Permiso != 1
BEGIN
	SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	RETURN
END	

DELETE Usuario WHERE Id = @IdUsuario
SET @Resultado = 'OK*Estimado Usuario el Usuario '+ @NombreUsuario +' se elimino correctamente'	

SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_IniciarSesion] @Usuario varchar(255), @Password varchar(255), 
@Resultado varchar(max) OUTPUT
AS
BEGIN
DECLARE @ExisteUser INT, @ValidaPassword varchar(255), @IdUser varchar(255),
	@PasswordMd5 varchar(255), @UsuarioMd5 varchar(255), @IdUsuario int, @Vigencia datetime,
	@NombreUser varchar(255), @FechaActual datetime, @IdEstado Int
	SELECT @UsuarioMd5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Usuario ), 2)
	SELECT @PasswordMd5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Password ), 2)
	SELECT @ExisteUser = COUNT(1) FROM Usuario Where IdUser = @UsuarioMd5
	SELECT @ValidaPassword = Password  FROM Usuario WHERE IdUser = @UsuarioMd5
	SELECT @IdUsuario = Id FROM Usuario Where IdUser = @UsuarioMd5
	SELECT @NombreUser = Usuario FROM Usuario WHERE IdUser =  @UsuarioMd5
	SELECT @Vigencia = FechaVigencia FROM Usuario WHERE IdUser = @UsuarioMd5
	SELECT @FechaActual = GETDATE()
	Select @IdEstado = IdEstado FROM Usuario WHERE Id = @IdUsuario
	
	IF @ExisteUser = 0
		BEGIN
			SET @Resultado = 'Error*El usuario ingresado no existe'
		END
	ELSE
		BEGIN		
		SELECT @IdUser = IdUser FROM Usuario where IdUser = @UsuarioMd5
			IF @ValidaPassword = @PasswordMd5
				BEGIN
					IF @FechaActual < @Vigencia  
						BEGIN
							IF @IdEstado = 1
								BEGIN
									INSERT INTO Log_InicioSesion (IdUser,FechaIngreso)VALUES(@IdUsuario, GETDATE())
									SET @Resultado = 'OK*'+@IdUser
								END
							ELSE
								BEGIN
									SET @Resultado = 'Error*El usuario se encuentra inactivo, por favor comuniquese con el administrador de la aplicación'
								END
						END
					ELSE
						BEGIN
							SET @Resultado = 'Error*El usuario '+@NombreUser+' no tiene permisos para ingresar a la aplicación, la fecha limite de ingreso se encuentra vencida. Si desea activar un nuevo periodo de tiempo comuniquese con el administrador de la aplicación.'
						END					
				END
			ELSE
				BEGIN
					SET @Resultado = 'Error*Contraseña incorrecta'
				END
		END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_UltimoIngresoUsuario] @IdUser varchar(255)
AS
BEGIN
	SELECT	TOP 1 CONVERT(varchar(255),FechaIngreso,105)+ ' ' + CONVERT(varchar(255),FechaIngreso,8) AS FechaUltimoIngresoUser
	FROM	(SELECT TOP 2 A.FechaIngreso
	FROM	Log_InicioSesion A,
			Usuario B
	WHERE	B.IdUser = @IdUser	
			AND B.Id = A.IdUser
			ORDER BY A.Id DESC) AS FechaUltimoIngresoUser 
	ORDER BY FechaIngreso ASC
END

GO

CREATE PROCEDURE [dbo].[SP_ListaMenu] @IdUsuario varchar(255)
AS
BEGIN
DECLARE @Id int
SELECT	@Id = dbo.Fun_BuscarIdUsuario(@IdUsuario)
	SELECT	A.Id, 
			'<i class="'+ A.Icono +'"' + 'style="font-size:' + A.SizeIcono + ';color:'+ A.Color+'"></i>' as IconoCodigoHtml,
			A.Nombre,
			A.Ruta
	FROM	Menu A,
			Permiso_Menu B
	WHERE	B.IdUsuario = @Id
			and B.Permiso = 1
			and A.IdEstado =1
			and A.Id = B.IdMenu
			order by Orden asc
END

CREATE PROCEDURE [dbo].[SP_CrearPermisoMenu] 
				@IdUser varchar(255),  
				@IdUsuarioMenu int,
				@IdMenu int,
				@PermisoMenu int,
				@Resultado varchar(255) OUTPUT     
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)
	
SET @Accion = 'Crear'
SET @SubMenu = '/Configuracion/Permiso_Menu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	BEGIN		
		INSERT INTO Permiso_Menu
		(IdMenu, IdUsuario, Permiso, CreateBy, DateCreate) 
		VALUES 
		(@IdMenu, @IdUsuarioMenu, @PermisoMenu, @IdUsuario, GETDATE()) 

		SET @Resultado = 'OK*El Permiso al Menu a sido creado correctamente'
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_ActualizarPermisoMenu]	@IdUser varchar(255),  
													@IdPermisoMenu int,
													@IdUsuarioMenu int,
													@IdMenu int,
													@PermisoMenu int,
													@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)
	
SET @Accion = 'Editar'
SET @SubMenu = '/Configuracion/Permiso_Menu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	IF(exists(select * from Permiso_Menu where IdMenu = @IdMenu AND IdUsuario = @IdUsuarioMenu AND Id != @IdPermisoMenu))  
	BEGIN  
		SET @Resultado = 'Error*El Permiso ya existe.'
		RETURN  
	END
ELSE
	BEGIN
		UPDATE	Permiso_Menu 
		SET		IdMenu = @IdMenu, 
				IdUsuario = @IdUsuarioMenu,
				Permiso = @PermisoMenu
		WHERE	Id = @IdPermisoMenu
		
		SET @Resultado = 'OK*El Permiso se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_EliminarPermisoMenu] @IdUser varchar(255), @IdPermisoMenu int, @Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@Accion varchar(255), 
		@SubMenu varchar(255)

SET @Accion = 'Eliminar'
SET @SubMenu = '/Configuracion/Permiso_Menu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT	@Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)	

IF @Permiso != 1
BEGIN
	SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	RETURN
END	

DELETE Permiso_Menu WHERE Id = @IdPermisoMenu
SET @Resultado = 'OK*Estimado Usuario el Permiso se elimino correctamente'	

SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_ListaSubMenu] @IdUsuario varchar(255), @Modulo varchar(255)
AS
BEGIN
DECLARE @Id int, @IdMenu int
SELECT @IdMenu = Id From Menu WHERE Ruta = @Modulo
SELECT	@Id = dbo.Fun_BuscarIdUsuario(@IdUsuario)
	SELECT	A.Id,	
			A.IdMenu,	
			'<i class="'+ A.Icono +'"' + 'style="font-size:' + A.SizeIcono + ';color:'+ A.Color+'"></i>' as IconoCodigoHtml, 
			A.Nombre, 
			A.Ruta,
			A.ColorFondo
	FROM	SubMenu A,
			Permiso_SubMenu B
	WHERE	B.IdUsuario = @Id
			and B.Permiso = 1
			and A.IdEstado =1 
			and A.IdMenu = @IdMenu
			and A.Id = B.IdSubMenu
END

GO

CREATE FUNCTION [dbo].[Fun_PermisoUsuario](@Accion varchar(255), @IdUsuario Int, @Ruta varchar(255))
RETURNS int
AS
BEGIN	
	DECLARE @Resultado Int
	
	IF @Accion = 'Crear'
	BEGIN
		SELECT	@Resultado = P.Crear
		FROM	Usuario U,
				Permiso_SubMenu P,
				SubMenu S
		WHERE	U.Id = P.IdUsuario  AND
				S.Id = P.IdSubMenu AND
				P.IdUsuario = @IdUsuario AND
				S.Ruta = @Ruta
	END
	ELSE IF @Accion = 'Editar'
		BEGIN
			SELECT	@Resultado = P.Editar
			FROM	Usuario U,
					Permiso_SubMenu P,
					SubMenu S
			WHERE	U.Id = P.IdUsuario  and
					S.Id = P.IdSubMenu AND
					P.IdUsuario = @IdUsuario AND
					S.Ruta = @Ruta
		END
	ELSE IF @Accion = 'Eliminar'
	BEGIN
		SELECT	@Resultado = P.Eliminar
		FROM	Usuario U,
				Permiso_SubMenu P,
				SubMenu S
		WHERE	U.Id = P.IdUsuario  and
				S.Id = P.IdSubMenu AND
				P.IdUsuario = @IdUsuario AND
				S.Ruta = @Ruta
	END
	RETURN @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_GridPermisoSubMenu]
AS
BEGIN
	SELECT
	A.Id,
	P.IdUsuario,
	U.Usuario,
	P.IdMenu,
	M.Nombre AS NombreMenu,
	A.IdSubMenu,
	S.Nombre AS NombreSubMenu,
	A.Permiso,
	CASE P.Permiso
	WHEN 1 THEN 'Activo'
	ELSE 'Inactivo'
	END AS PermisoTexto,

	A.Crear,
	CASE A.Crear
	WHEN 1 THEN 'Activo'
	ELSE 'Inactivo'
	END AS CrearTexto,

	A.Editar,
	CASE A.Editar
	WHEN 1 THEN 'Activo'
	ELSE 'Inactivo'
	END AS EditarTexto,

	A.Eliminar,
	CASE A.Eliminar
	WHEN 1 THEN 'Activo'
	ELSE 'Inactivo'
	END AS EliminarTexto,

	dbo.Fun_BuscarNombreUsuario(P.CreateBy) As CreateBy,
			CONVERT(varchar(255),P.DateCreate,105)+ ' ' + CONVERT(varchar(255),P.DateCreate,8) As DateCreate
	FROM	Permiso_Menu P,
			Usuario U,
			Menu M,	
			SubMenu S,
			Permiso_SubMenu A
	WHERE	P.IdUsuario = U.Id AND
			M.Id = P.IdMenu AND
			S.IdMenu = M.Id AND
			S.Id = A.IdSubMenu AND
			U.Id = A.IdUsuario
END

GO

CREATE PROCEDURE [dbo].[SP_CrearPermisoSubMenu] 
				@IdUser varchar(255),  
				@IdUsuarioSubMenu int,
				@IdSubMenu int,
				@Ver int,
				@Crear int,
				@Editar int,
				@Eliminar int,
				@Resultado varchar(255) OUTPUT     
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255),
		@IdMenu INT,
		@Validar INT
	
SET @Accion = 'Crear'
SET @SubMenu = '/Configuracion/Permiso_SubMenu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)
SELECT @IdMenu = IdMenu FROM SubMenu WHERE Id = @IdSubMenu 
SELECT @Validar = COUNT(1) 

IF @Permiso > 0
	BEGIN		
	IF(exists(select * from Permiso_Menu where IdMenu = @IdMenu AND IdUsuario = @IdUsuarioSubMenu))  
		BEGIN  
			INSERT INTO Permiso_SubMenu
			(IdSubMenu, IdUsuario, Permiso, Crear, Editar, Eliminar, CreateBy, DateCreate) 
			VALUES 
			(@IdSubMenu, @IdUsuarioSubMenu, @Ver, @Crear, @Editar, @Eliminar, @IdUsuario, GETDATE()) 
			SET @Resultado = 'OK*El Permiso al Menu a sido creado correctamente'
			RETURN
		END
	ELSE
		BEGIN
			INSERT INTO Permiso_Menu (IdMenu, IdUsuario, CreateBy, DateCreate) VALUES (@IdMenu, @IdUsuarioSubMenu, @IdUsuario, GETDATE())

			INSERT INTO Permiso_SubMenu
			(IdSubMenu, IdUsuario, Permiso, Crear, Editar, Eliminar, CreateBy, DateCreate) 
			VALUES 
			(@IdSubMenu, @IdUsuarioSubMenu, @Ver, @Crear, @Editar, @Eliminar, @IdUsuario, GETDATE()) 

			SET @Resultado = 'OK*El Permiso al Menu a sido creado correctamente'
		END
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_ActualizarPermisoSubMenu]	@IdUser varchar(255),  
														@IdPermisoSubMenu int,
														@IdUsuarioSubMenu int,
														@IdSubMenu int,
														@Ver int,
														@Crear int,
														@Editar int,
														@Eliminar int,
														@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)
	
SET @Accion = 'Editar'
SET @SubMenu = '/Configuracion/Permiso_SubMenu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	IF(exists(select * from Permiso_SubMenu where IdSubMenu = @IdSubMenu AND IdUsuario = @IdUsuarioSubMenu AND Id != @IdPermisoSubMenu))  
	BEGIN  
		SET @Resultado = 'Error*El Permiso ya existe.'
		RETURN  
	END
ELSE
	BEGIN
		UPDATE	Permiso_SubMenu 
		SET		IdSubMenu = @IdSubMenu, 
				IdUsuario = @IdUsuarioSubMenu,
				Permiso = @Ver,
				Crear = @Crear,
				Editar = @Editar,
				Eliminar = @Eliminar
		WHERE	Id = @IdPermisoSubMenu
		
		SET @Resultado = 'OK*El Permiso se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_EliminarPermisoSubMenu] @IdUser varchar(255), @IdPermisoSubMenu int, @Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@Accion varchar(255), 
		@SubMenu varchar(255)

SET @Accion = 'Eliminar'
SET @SubMenu = '/Configuracion/Permiso_SubMenu'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT	@Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)	

IF @Permiso != 1
BEGIN
	SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	RETURN
END	

DELETE Permiso_SubMenu WHERE Id = @IdPermisoSubMenu
SET @Resultado = 'OK*Estimado Usuario el Permiso se elimino correctamente'	

SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_GridEmpresa]
AS
BEGIN
	SELECT	Id, 
			Nombre, 
			IdTipoDocumento,
			dbo.Fun_BuscarNombreTipoDocumento(IdTipoDocumento) As TipoDocumento,
			Identificacion,
			Email,
			Telefono,
			Contacto,
			Direccion,
			IdCiudad,
			dbo.Fun_BuscarNombreCiudad(IdCiudad) As Ciudad,
			Logo,
			IdEstado,
			dbo.Fun_BuscarNombreTipoEstado(IdEstado) As Estado,
			dbo.Fun_BuscarNombreUsuario(CreateBy) As CreateBy,
			CONVERT(varchar(255),DateCreate,105)+ ' ' + CONVERT(varchar(255),DateCreate,8) As DateCreate			
			FROM Empresa
			ORDER BY Nombre Desc
END

													@IdEstado int,

AS
BEGIN

		@IdUsuario INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)

CREATE PROCEDURE [dbo].[SP_CrearCLE] 
				@IdUser varchar(255), 			
				@IdEmpleado int,
				@IdEmpresa int,
				@IdCargo int,
				@IdTipoContrato int,
				@SalarioMensual int,

				@FechaEnd varchar(255), 
				@IdEps INT,
				@PorcentajeContEps DECIMAL(10,2),
				@IdFondoPension INT, 
				@PorcentajeContFP DECIMAL(10,2), 
				@IdBanco INT,
				@NumeroCuentaPago VARCHAR(255), 
				@SubTransporte int,
				@IdCesantias INT,
				@Observacion varchar(255), 	
				@Resultado varchar(MAX) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@ExisteCLEActivo INT,
		@IdContratoLaboral INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255),
		@FechaInicio varchar(255) = CONVERT(VARCHAR(255),DATEPART(YEAR,@FechaIni))+'-'+CONVERT(VARCHAR(255),DATEPART(MONTH,@FechaIni))+'-'+CONVERT(VARCHAR(255),DATEPART(DAY,@FechaIni)),
		@FechaFin varchar(255) = CONVERT(VARCHAR(255),DATEPART(YEAR,@FechaEnd))+'-'+CONVERT(VARCHAR(255),DATEPART(MONTH,@FechaEnd))+'-'+CONVERT(VARCHAR(255),DATEPART(DAY,@FechaEnd))
		
	
SET @Accion = 'Crear'
SET @SubMenu = '/Recursos_Humanos/Contrato_Laboral'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

SELECT @ExisteCLEActivo = COUNT(1) FROM ContratoLaboral WHERE IdEmpleado = @IdEmpleado AND IdEstado = 1

IF @Permiso > 0
	BEGIN
		IF @ExisteCLEActivo = 0
			BEGIN
				INSERT INTO ContratoLaboral
				(IdEmpleado, IdEmpresa, IdCargo, IdTipoContrato, SalarioMensual, FechaInicio,FechaFin, IdEps, PorcentajeEps, IdFondoPension, PorcentajeFondoPension,
				IdBanco, NumeroCuenta,SubTransporte,IdFondoCesantias,Observacion, CreateBy, DateCreate) 
				VALUES 
				(@IdEmpleado, @IdEmpresa, @IdCargo, @IdTipoContrato, @SalarioMensual, @FechaInicio, @FechaFin, @IdEps, @PorcentajeContEps,
				@IdFondoPension, @PorcentajeContFP,@IdBanco,@NumeroCuentaPago, @SubTransporte, @IdCesantias,@Observacion, @IdUsuario, GETDATE()) 
											
				SET @Resultado = 'OK*El Contrato Laboral para el Empleado a sido creado correctamente'
			END													
		ELSE
			BEGIN
				SET @Resultado = 'Error*El Empleado ya cuenta con un Contrato Laboral Activo, solo se permite un contrato Laboral Activo'
			END
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_ActualizarCLE]	@IdUser varchar(255),
											@IdCLE int,	
											@IdCargo int,	
											@IdTipoContrato int,
											@SalarioMensual int,
											@FechaEnd varchar(255),
											@IdEps INT,
											@PorcentajeContEps DECIMAL(10,2),
											@IdFondoPension INT, 
											@PorcentajeContFP DECIMAL(10,2), 
											@IdBanco INT,
											@NumeroCuentaPago VARCHAR(255), 
											@SubTransporte int,
											@IdCesantias INT,
											@Observacion varchar(255),
											@IdEstado int,			
											@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255),
		@Existe int,
		@IdEmpleado int,
		@FechaFin varchar(255) = CONVERT(VARCHAR(255),DATEPART(YEAR,@FechaEnd))+'-'+CONVERT(VARCHAR(255),DATEPART(MONTH,@FechaEnd))+'-'+CONVERT(VARCHAR(255),DATEPART(DAY,@FechaEnd))

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)
SELECT @IdEmpleado = IdEmpleado FROM ContratoLaboral WHERE Id = @IdCLE

SELECT @Existe = COUNT(1) FROM ContratoLaboral WHERE IdEmpleado = @IdEmpleado AND IdEstado = 1

DECLARE @IdCLEValida int 

SELECT @IdCLEValida = Id FROM ContratoLaboral where IdEmpleado = @IdEmpleado and IdEstado = 1

IF @Permiso > 0
	IF @IdEstado =  1 AND  @Existe != 0 AND @IdCLEValida != @IdCLE
	BEGIN  
		SET @Resultado = 'Error*El Empleado tiene un Contrato Laboral ya vigente, solo puede tener 1 Contrato Laboral ACTIVO a la vez.'
		RETURN  
	END
ELSE
	BEGIN
		UPDATE	ContratoLaboral
			SET		IdCargo = @IdCargo,
					IdTipoContrato = @IdTipoContrato,
					SalarioMensual = @SalarioMensual,
					FechaFin = @FechaFin,
					IdEps = @IdEps,
					PorcentajeEps = @PorcentajeContEps,
					IdFondoPension = @IdFondoPension,
					PorcentajeFondoPension = @PorcentajeContFP,
					IdBanco = @IdBanco,
					NumeroCuenta = @NumeroCuentaPago,
					SubTransporte = @SubTransporte,
					IdFondoCesantias = @IdCesantias,
					Observacion = @Observacion,
					IdEstado = @IdEstado
			WHERE	Id = @IdCLE
		
		SET @Resultado = 'OK*El Contrato Laboral se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado

CREATE PROCEDURE [dbo].[SP_GridCLE]
AS
BEGIN
SELECT	A.Id,	 
			A.IdEmpleado,
			dbo.Fun_BuscarNombreEmpleado(D.Id) AS Empleado,	
			C.Id AS IdEmpresa,
			dbo.Fun_BuscarNombreEmpresa(C.Id) AS Empresa,						
			A.IdCargo,
			dbo.Fun_BuscarNombreCargo(A.IdCargo) AS Cargo,
			A.IdTipoContrato,
			dbo.Fun_BuscarNombreTipoContrato(A.IdTipoContrato) AS Contrato,
			A.SalarioMensual,
			A.IdEps,
			A.IdFondoPension,
			A.IdFondoCesantias,
			A.IdBanco,
			A.PorcentajeEps,
			A.PorcentajeFondoPension,
			A.NumeroCuenta,
			A.SubTransporte,
			'$ ' + CONVERT(varchar(255),A.SalarioMensual) As TextoSalarioMensual,
			CONVERT(varchar(255),A.FechaInicio,23) As FechaInicio,
			CONVERT(varchar(255), DATEPART(DAY, A.FechaInicio))+'/'+ CONVERT(varchar(255),DATEPART(MONTH, A.FechaInicio))+'/'+ CONVERT(varchar(255),DATEPART(YEAR, A.FechaInicio)) As TextoFechaInicio,
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN '0000-00-00'
			ELSE CONVERT(varchar(255),A.FechaFin,23)
			END as FechaFin,			
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN 'Por Definir'
			ELSE CONVERT(varchar(255), DATEPART(DAY, A.FechaFin))+'/'+ CONVERT(varchar(255),DATEPART(MONTH, A.FechaFin))+'/'+ CONVERT(varchar(255),DATEPART(YEAR, A.FechaFin))						
			END as TextoFechaFin,
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN CONVERT(varchar(255),DATEDIFF(DAY,A.FechaInicio,GETDATE())+1) + ' Día(s)'
			ELSE CONVERT(varchar(255),DATEDIFF(DAY,A.FechaInicio,A.FechaFin)+1) + ' Día(s)'
			END AS Permanencia,			
			A.Observacion,
			A.IdEstado,
			dbo.Fun_BuscarNombreTipoEstado(A.IdEstado) As Estado,
			dbo.Fun_BuscarNombreUsuario(A.CreateBy) As CreateBy,
			CONVERT(varchar(255),A.DateCreate,105)+ ' ' + CONVERT(varchar(255),A.DateCreate,8) As DateCreate,		
			D.Imagen
			FROM ContratoLaboral A,
			Empresa C,
			Empleado D
	WHERE	A.IdEmpresa = C.Id
			AND D.Id = A.IdEmpleado
			ORDER BY Id Desc
END

GO

CREATE PROCEDURE [dbo].[SP_ListaContratoLaboral]
AS
BEGIN
	SELECT A.Id, A.Nombre + ' ' + A.Apellidos As Nombre 
	FROM	Empleado A,
			ContratoLaboral B	
	WHERE	A.Id = B.IdEmpleado AND
			A.IdEstado = 1 AND
			B.IdEstado = 1 ORDER BY Nombre ASC
END

BEGIN

		@IdUsuario INT, 
		@ExisteRegistroTablaPrincipal INT,
		@FechaInicio varchar(255) = CONVERT(VARCHAR(255),DATEPART(YEAR,@FechaIni))+'-'+CONVERT(VARCHAR(255),DATEPART(MONTH,@FechaIni))+'-'+CONVERT(VARCHAR(255),DATEPART(DAY,@FechaIni)),

		IF @ExisteRegistroTablaPrincipal = 0
			BEGIN
				INSERT INTO SucursalEmpleado

				VALUES 
				(@IdEmpleado, @IdSucursal, @FechaInicio, @FechaFin, @Observacion, @IdUsuario, GETDATE())

												@Observacion varchar(255),

												@Resultado varchar(255) OUTPUT

BEGIN

		@IdUsuario INT,
		@Existe INT,

	BEGIN

CREATE PROCEDURE [dbo].[SP_GridSucursalEmpleado]
AS
BEGIN
SELECT	A.Id,	 
			C.Id AS IdEmpresa,
			dbo.Fun_BuscarNombreEmpresa(C.Id) AS Empresa,
			B.Id AS IdSucursal,
			dbo.Fun_BuscarNombreSucursal(B.Id) AS Sucursal,
			A.IdEmpleado,
			dbo.Fun_BuscarNombreEmpleado(A.IdEmpleado) AS Empleado,	
			CONVERT(varchar(255),A.FechaInicio,23) As FechaInicio,
			CONVERT(varchar(255),A.FechaInicio,105) As TextoFechaInicio,
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN '0000-00-00'
			ELSE CONVERT(varchar(255),A.FechaFin,23)
			END as FechaFin,			
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN 'Por Definir'
			ELSE CONVERT(varchar(255),A.FechaFin,105)			
			END as TextoFechaFin,
			CASE A.FechaFin
			WHEN '1900-01-01 00:00:00.000' THEN CONVERT(varchar(255),DATEDIFF(DAY,A.FechaInicio,GETDATE())+1) + ' Día(s)'
			ELSE CONVERT(varchar(255),DATEDIFF(DAY,A.FechaInicio,A.FechaFin)+1) + ' Día(s)'
			END AS Permanencia,			
			A.Observacion,
			A.IdEstado,
			dbo.Fun_BuscarNombreTipoEstado(A.IdEstado) As Estado,
			dbo.Fun_BuscarNombreUsuario(A.CreateBy) As CreateBy,
			CONVERT(varchar(255),A.DateCreate,105)+ ' ' + CONVERT(varchar(255),A.DateCreate,8) As DateCreate,			
			D.Imagen
			FROM SucursalEmpleado A,
			Sucursal B,
			Empresa C,
			Empleado D
	WHERE	A.IdSucursal = B.Id
			AND B.IdEmpresa = C.Id
			AND D.Id = A.IdEmpleado
			ORDER BY Empleado Desc
END

GO
