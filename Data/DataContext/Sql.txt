CREATE PROCEDURE [dbo].[SP_IniciarSesion] @Usuario varchar(255), @Password varchar(255), 
@Resultado varchar(max) OUTPUT
AS
BEGIN
DECLARE @ExisteUser INT, @ValidaPassword varchar(255), @IdUser varchar(255),
	@PasswordMd5 varchar(255), @UsuarioMd5 varchar(255), @IdUsuario int, @Vigencia datetime,
	@NombreUser varchar(255), @FechaActual datetime, @IdEstado Int
	SELECT @UsuarioMd5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Usuario ), 2)
	SELECT @PasswordMd5 = CONVERT ( VARCHAR (32), HashBytes ( 'MD5' ,  @Password ), 2)
	SELECT @ExisteUser = COUNT(1) FROM Usuario Where IdUser = @UsuarioMd5
	SELECT @ValidaPassword = Password  FROM Usuario WHERE IdUser = @UsuarioMd5
	SELECT @IdUsuario = Id FROM Usuario Where IdUser = @UsuarioMd5
	SELECT @NombreUser = Usuario FROM Usuario WHERE IdUser =  @UsuarioMd5
	SELECT @Vigencia = FechaVigencia FROM Usuario WHERE IdUser = @UsuarioMd5
	SELECT @FechaActual = GETDATE()
	Select @IdEstado = IdEstado FROM Usuario WHERE Id = @IdUsuario
	
	IF @ExisteUser = 0
		BEGIN
			SET @Resultado = 'Error*El usuario ingresado no existe'
		END
	ELSE
		BEGIN		
		SELECT @IdUser = IdUser FROM Usuario where IdUser = @UsuarioMd5
			IF @ValidaPassword = @PasswordMd5
				BEGIN
					IF @FechaActual < @Vigencia  
						BEGIN
							IF @IdEstado = 1
								BEGIN
									INSERT INTO Log_InicioSesion (IdUser,FechaIngreso)VALUES(@IdUsuario, GETDATE())
									SET @Resultado = 'OK*'+@IdUser
								END
							ELSE
								BEGIN
									SET @Resultado = 'Error*El usuario se encuentra inactivo, por favor comuniquese con el administrador de la aplicación'
								END
						END
					ELSE
						BEGIN
							SET @Resultado = 'Error*El usuario '+@NombreUser+' no tiene permisos para ingresar a la aplicación, la fecha limite de ingreso se encuentra vencida. Si desea activar un nuevo periodo de tiempo comuniquese con el administrador de la aplicación.'
						END					
				END
			ELSE
				BEGIN
					SET @Resultado = 'Error*Contraseña incorrecta'
				END
		END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_UltimoIngresoUsuario] @IdUser varchar(255)
AS
BEGIN
	SELECT	TOP 1 CONVERT(varchar(255),FechaIngreso,100) AS FechaUltimoIngresoUser
	FROM	(SELECT TOP 2 A.FechaIngreso
	FROM	Log_InicioSesion A,
			Usuario B
	WHERE	B.IdUser = @IdUser	
			AND B.Id = A.IdUser
			ORDER BY A.Id DESC) AS FechaUltimoIngresoUser 
	ORDER BY FechaIngreso ASC
END

GO

AS
BEGIN
DECLARE @Id int
SELECT	@Id = dbo.Fun_BuscarIdUsuario(@IdUsuario)
	SELECT	A.Id, 
			'<i class="'+ A.Icono +'"' + 'style="font-size:' + A.SizeIcono + ';color:'+ A.Color+'"></i>' as IconoCodigoHtml,
			A.Nombre,
			A.Ruta

			Permiso_Menu B
	WHERE	B.IdUsuario = @Id
			and B.Permiso = 1
			and A.IdEstado =1
			and A.Id = B.IdMenu

CREATE PROCEDURE [dbo].[SP_ListaSubMenu] @IdUsuario varchar(255), @Modulo varchar(255)
AS
BEGIN
DECLARE @Id int, @IdMenu int
SELECT @IdMenu = Id From Menu WHERE Ruta = @Modulo
SELECT	@Id = dbo.Fun_BuscarIdUsuario(@IdUsuario)
	SELECT	A.Id,	
			A.IdMenu,	
			'<i class="'+ A.Icono +'"' + 'style="font-size:' + A.SizeIcono + ';color:'+ A.Color+'"></i>' as IconoCodigoHtml, 
			A.Nombre, 
			A.Ruta
	FROM	SubMenu A,
			Permiso_SubMenu B
	WHERE	B.IdUsuario = @Id
			and B.Permiso = 1
			and A.IdEstado =1 
			and A.IdMenu = @IdMenu
			and A.Id = B.IdSubMenu
END

GO

CREATE FUNCTION [dbo].[Fun_PermisoUsuario](@Accion varchar(255), @IdUsuario Int, @Ruta varchar(255))
RETURNS int
AS
BEGIN	
	DECLARE @Resultado Int
	
	IF @Accion = 'Crear'
	BEGIN
		SELECT	@Resultado = P.Crear
		FROM	Usuario U,
				Permiso_SubMenu P,
				SubMenu S
		WHERE	U.Id = P.IdUsuario  and
				S.Id = P.IdSubMenu AND
				S.Ruta = @Ruta
	END
	ELSE IF @Accion = 'Editar'
		BEGIN
			SELECT	@Resultado = P.Editar
			FROM	Usuario U,
					Permiso_SubMenu P,
					SubMenu S
			WHERE	U.Id = P.IdUsuario  and
					S.Id = P.IdSubMenu AND
					S.Ruta = @Ruta
		END
	ELSE IF @Accion = 'Eliminar'
	BEGIN
		SELECT	@Resultado = P.Eliminar
		FROM	Usuario U,
				Permiso_SubMenu P,
				SubMenu S
		WHERE	U.Id = P.IdUsuario  and
				S.Id = P.IdSubMenu AND
				S.Ruta = @Ruta
	END
	RETURN @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_ActualizarEmpresa]	@IdUser varchar(255),
												@IdEmpresa int, 				
												@NombreEmpresa varchar(255), 
												@IdTipoDocumento int,
												@Identificacion varchar(255),
												@Email varchar(255), 
												@Telefono varchar(255),  
												@Contacto varchar(255),
												@IdCiudad int,
												@Direccion varchar(255),
												@IdEstado int,			
												@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@Accion varchar(255), 
		@SubMenu varchar(255),

		@NombreEmpresav varchar(255) = (SELECT Nombre FROM Empresa WHERE Id = @IdEmpresa),
		@IdTipoDocumentov varchar(255) = (SELECT IdTipoDocumento FROM Empresa WHERE Id = @IdEmpresa),
		@Identificacionv varchar(255) = (SELECT Identificacion FROM Empresa WHERE Id = @IdEmpresa),
		@Emailv varchar(255) = (SELECT Email FROM Empresa WHERE Id = @IdEmpresa),
		@Telefonov varchar(255) = (SELECT Telefono FROM Empresa WHERE Id = @IdEmpresa),
		@Contactov varchar(255) = (SELECT Contacto FROM Empresa WHERE Id = @IdEmpresa),
		@Direccionv varchar(255) = (SELECT Direccion FROM Empresa WHERE Id = @IdEmpresa),
		@IdCiudadv varchar(255) = (SELECT IdCiudad FROM Empresa WHERE Id = @IdEmpresa),
		@IdEstadov varchar(255) = (SELECT IdEstado FROM Empresa WHERE Id = @IdEmpresa)	 

SET @Accion = 'Crear'
SET @SubMenu = '/Negocio/Empresa'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	IF(exists(select * from Empresa where Nombre = @NombreEmpresa AND IdTipoDocumento = @IdTipoDocumento AND Identificacion = @Identificacion AND Id != @IdEmpresa))  
	BEGIN  
		SET @Resultado = 'Error*La Empresa ' + @NombreEmpresa +' ya existe.'
		RETURN  
	END
ELSE
	BEGIN
		UPDATE	Empresa 
			SET		Nombre = @NombreEmpresa, 
					IdTipoDocumento =  @IdTipoDocumento,
					Identificacion = @Identificacion,								 
					Email = @Email, 
					Telefono = @Telefono,
					Contacto = @Contacto,
					IdCiudad = @IdCiudad,
					Direccion = @Direccion,
					IdEstado = @IdEstado
					WHERE Id = @IdEmpresa

		INSERT INTO LogsAcciones 
		(IdUsuario, Accion, Descripcion, Fecha) 
		VALUES 
		(@IdUsuario, 
		'Editar', 
		'Nombres:'  + (SELECT CASE WHEN @NombreEmpresav != @NombreEmpresa THEN '(N) '+@NombreEmpresa+' - (A) '+ @NombreEmpresav ELSE 'No' END) +
		', Tipo Doc:' + (SELECT CASE WHEN @IdTipoDocumentov != @IdTipoDocumento THEN '(N) '+dbo.Fun_BuscarNombreTipoDocumento(@IdTipoDocumento)+' - (A) '+ dbo.Fun_BuscarNombreTipoDocumento(@IdTipoDocumentov) ELSE 'No' END) +
		', Identificación:' + (SELECT CASE WHEN @Identificacionv != @Identificacion THEN '(N) '+@Identificacion+' - (A) '+ @Identificacionv ELSE 'No' END) +
		', Email:'+ (SELECT CASE WHEN @Emailv != @Email THEN '(N) '+@Email+' - (A) '+ @Emailv ELSE 'No' END) +
		', Telefono:'+ (SELECT CASE WHEN @Telefonov != @Telefono THEN '(N) '+@Telefono+' - (A) '+ @Telefonov ELSE 'No' END) +
		', Contacto:'+ (SELECT CASE WHEN @Contactov != @Contacto THEN '(N) '+@Contacto+' - (A) '+ @Contactov ELSE 'No' END) +
		', Dirección:'+ (SELECT CASE WHEN @Direccionv != @Direccion THEN '(N) '+@Direccion+' - (A) '+ @Direccionv ELSE 'No' END) +
		', Ciudad:'+ (SELECT CASE WHEN @IdCiudadv != @IdCiudad THEN '(N) '+dbo.Fun_BuscarNombreCiudad(@IdCiudad)+' - (A) '+ dbo.Fun_BuscarNombreCiudad(@IdCiudadv) ELSE 'No' END) +
		', Estado:'+ (SELECT CASE WHEN @IdEstadov != @IdEstado THEN '(N) '+dbo.Fun_BuscarNombreTipoEstado(@IdEstado)+' - (A) '+ dbo.Fun_BuscarNombreTipoEstado(@IdEstadov) ELSE 'No' END), 
		GETDATE())		

		SET @Resultado = 'OK*La Empresa ' + @NombreEmpresa + ' se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

												@IdEmpleado int,				 
												@Nombre varchar(255),
												@Apellidos varchar(255),
												@IdTipoDocumento int,
												@Identificacion varchar(255),
												@IdEstado int,			
												@Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT,
		@Accion VARCHAR(255),
		@SubMenu VARCHAR(255)
	
SET @Accion = 'Editar'
SET @SubMenu = '/Recursos_Humanos/Empleado'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)

IF @Permiso > 0
	IF(exists(select * from Empleado where IdTipoDocumento = @IdTipoDocumento AND Identificacion = @Identificacion AND Id != @IdEmpleado))  
	BEGIN  
		SET @Resultado = 'OK*El Empleado ' +  @Nombre + ' '+ @Apellidos +' ya existe.'
		RETURN  
	END
ELSE
	BEGIN
		UPDATE	Empleado 
		SET		Nombre = @Nombre, 
				Apellidos = @Apellidos,
				IdTipoDocumento = @IdTipoDocumento,
				Identificacion = @Identificacion,
				IdEstado = @IdEstado
		WHERE	Id = @IdEmpleado
		
		SET @Resultado = 'OK*El Empleado ' +  @Nombre + ' '+ @Apellidos + ' se actualizo correctamente'  
	END	
ELSE
	BEGIN
		SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	END
	SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_EliminarEmpleado] @IdUser varchar(255), @IdEmpleado int, @Resultado varchar(255) OUTPUT
AS
BEGIN
DECLARE @Permiso INT, 
		@IdUsuario INT, 
		@NombreEmpleado varchar(255),
		@ApellidoEmpleado varchar(255),
		@Accion varchar(255), 
		@SubMenu varchar(255)

SET @Accion = 'Eliminar'
SET @SubMenu = '/Recursos_Humanos/Empleado'

SELECT @IdUsuario = Id FROM Usuario WHERE IdUser = @IdUser
SELECT @NombreEmpleado = Nombre FROM Empleado WHERE Id = @IdEmpleado
SELECT @ApellidoEmpleado = Apellidos FROM Empleado WHERE Id = @IdEmpleado
SELECT	@Permiso = dbo.Fun_PermisoUsuario(@Accion, @IdUsuario, @SubMenu)	

IF @Permiso != 1
BEGIN
	SET @Resultado = 'Error*Estimado Usuario usted no tiene permisos para realizar esta Acción.'
	RETURN
END	

DELETE Empleado WHERE Id = @IdEmpleado
SET @Resultado = 'OK*Estimado Usuario el Empleqado '+ @NombreEmpleado +' ' + @ApellidoEmpleado +' se elimino correctamente'	

SELECT @Resultado
END

GO

CREATE PROCEDURE [dbo].[SP_GridEmpleado]
AS
BEGIN
	SELECT	Id, 
			Imagen,
			Nombre,
			Apellidos,
			IdTipoDocumento,
			dbo.Fun_BuscarNombreTipoDocumento(IdTipoDocumento) As TipoDocumento,			
			Identificacion,
			IdEstado,
			dbo.Fun_BuscarNombreTipoEstado(IdEstado) As Estado,			
			dbo.Fun_BuscarNombreUsuario(CreateBy) As CreateBy,
			CONVERT(varchar(255), DATEPART(DAY, DateCreate))+'/'+ CONVERT(varchar(255),DATEPART(MONTH, DateCreate))+'/'+ CONVERT(varchar(255),DATEPART(YEAR, DateCreate))+' '+ CONVERT(varchar(255),DATEPART(HOUR, DateCreate))+':'+ CONVERT(varchar(255),DATEPART(MINUTE, DateCreate)) As DateCreate
			FROM Empleado
			ORDER BY Id Desc
END
